cmake_minimum_required(VERSION 3.24)
include(cmake/CheckArithmeticFlags.cmake)
nr_check_arithmetic_flags()
project(opendlss_cuda_compat VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  message(FATAL_ERROR "The CUDA compatibility port supports Linux x86-64 only")
endif()
set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "Only the tested SM75 target is supported")
if(NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "75")
  message(FATAL_ERROR "This port requires CMAKE_CUDA_ARCHITECTURES=75")
endif()
# The receipt's host identity must also describe nvcc's host pass.
if(NOT "${CMAKE_CXX_COMPILER_ARG1}" STREQUAL "")
  message(FATAL_ERROR "CXX compiler command suffixes are unsupported; use compiler flags separately")
endif()
if(NOT CMAKE_CUDA_HOST_COMPILER)
  if(NOT "$ENV{CUDAHOSTCXX}" STREQUAL "")
    set(CMAKE_CUDA_HOST_COMPILER "$ENV{CUDAHOSTCXX}")
  else()
    set(CMAKE_CUDA_HOST_COMPILER "${CMAKE_CXX_COMPILER}")
  endif()
endif()
if(IS_ABSOLUTE "${CMAKE_CUDA_HOST_COMPILER}")
  set(nr_cuda_host "${CMAKE_CUDA_HOST_COMPILER}")
else()
  find_program(nr_cuda_host NAMES "${CMAKE_CUDA_HOST_COMPILER}" REQUIRED NO_CACHE)
endif()
if(NOT nr_cuda_host STREQUAL CMAKE_CXX_COMPILER)
  message(FATAL_ERROR "CUDA host compiler must match the CXX compiler")
endif()
set(CMAKE_CUDA_HOST_COMPILER "${nr_cuda_host}")
enable_language(CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(CUDAToolkit 12.6 REQUIRED)
find_package(Threads REQUIRED)
nr_check_arithmetic_flags()

# This list is also the native source identity. Changes to any member trigger
# reconfiguration; the generated identity header is excluded from its own hash.
set(native_sources
  src/runtime.cu src/kernels/numerics/numerics.cuh src/browser_numerics.cuh
  src/f16_compat.cuh src/attention_f13.cuh src/f13_dot.cuh src/gemm_f13.cuh
  src/window_f13_fused.cuh src/window_common.cuh
  include/nr_cuda.h CMakeLists.txt cmake/CheckArithmeticFlags.cmake cmake/CheckedCudaCompile.cmake
  toolchain/nr_build_info.h.in)
set(source_inventory "")
foreach(path IN LISTS native_sources)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" digest)
  string(APPEND source_inventory "${digest}  ${path}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
string(SHA256 NR_SOURCE_SHA256 "${source_inventory}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-source-sha256.txt" "${source_inventory}")
set(config_text "source=${NR_SOURCE_SHA256};cuda=${CMAKE_CUDA_COMPILER_ID}-${CMAKE_CUDA_COMPILER_VERSION};host=${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION};host_path=${CMAKE_CUDA_HOST_COMPILER};arch=75;standard=17;build=${CMAKE_BUILD_TYPE};cuda_flags=${CMAKE_CUDA_FLAGS};cxx_flags=${CMAKE_CXX_FLAGS};cuda_arg1=${CMAKE_CUDA_COMPILER_ARG1};cxx_arg1=${CMAKE_CXX_COMPILER_ARG1};nvcc_prepend=$ENV{NVCC_PREPEND_FLAGS};nvcc_append=$ENV{NVCC_APPEND_FLAGS};lineinfo;ftz=false")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  string(APPEND config_text ";cuda_${config}=${CMAKE_CUDA_FLAGS_${config}};cxx_${config}=${CMAKE_CXX_FLAGS_${config}}")
endforeach()
string(SHA256 NR_CONFIGURATION_SHA256 "${config_text}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-build-config.txt" "${config_text}\n")
configure_file(toolchain/nr_build_info.h.in generated/nr_build_info.h @ONLY)
foreach(variable NVCC_PREPEND_FLAGS NVCC_APPEND_FLAGS)
  string(SHA256 NR_${variable}_SHA256 "$ENV{${variable}}")
endforeach()
set(nr_cuda_launcher "${CMAKE_COMMAND};-DNR_EXPECTED_NVCC_PREPEND_FLAGS_SHA256=${NR_NVCC_PREPEND_FLAGS_SHA256};-DNR_EXPECTED_NVCC_APPEND_FLAGS_SHA256=${NR_NVCC_APPEND_FLAGS_SHA256};-P;${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckedCudaCompile.cmake;--")
if(CMAKE_CUDA_COMPILER_LAUNCHER)
  list(APPEND nr_cuda_launcher ${CMAKE_CUDA_COMPILER_LAUNCHER})
endif()
set(CMAKE_CUDA_COMPILER_LAUNCHER "${nr_cuda_launcher}")

add_library(nr_cuda SHARED src/runtime.cu)
target_include_directories(nr_cuda PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(nr_cuda PRIVATE CUDA::cudart Threads::Threads)
target_compile_options(nr_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--ftz=false>)
set_target_properties(nr_cuda PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
install(TARGETS nr_cuda LIBRARY DESTINATION lib)
install(FILES include/nr_cuda.h DESTINATION include)
